import calliopemini as CM
import neopixel

pixel = neopixel.NeoPixel(CM.pin_RGB, 3) 

def bounce(delay): 
    for no in range(0,2,1): 
        pixel[no] = (0, 0, 128)
        pixel.show()
        CM.sleep(delay)
        pixel[no] = (0, 0, 0) 

    for no in range(2,0,-1):
        pixel[no] = (0, 0, 128)
        pixel.show()
        CM.sleep(delay)
        pixel[no] = (0, 0, 0)         
 
while not CM.button_b.was_pressed():
    bounce(250)
